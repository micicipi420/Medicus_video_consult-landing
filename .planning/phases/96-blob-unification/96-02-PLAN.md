    test.describe('blob correlated motion (BR-02)', () => {
      test('max angular separation across 4 sublayers stays <= 8px under fast sweep', async ({ page }) => {
        await page.goto('/');
        // Wait for engine to start in cursor mode.
        await page.waitForFunction(() =&gt;
          (window as any).__blobDebug &amp;&amp; (window as any).__blobDebug.mode === 'cursor');

        // Reset rolling max.
        await page.evaluate(() =&gt; (window as any).__blobDebug.resetMaxSeparation());

        // Drive a fast diagonal sweep across the viewport in 30 steps.
        // 1280x800 viewport => sweep from (50, 50) to (1230, 750) = ~1377px / ~1.5s
        // = ~918 px/s — within VELOCITY_MAX (1500 px/s).
        const steps = 30;
        const dt = 50; // 30 * 50ms = 1500ms
        for (let i = 0; i &lt;= steps; i++) {
          const t = i / steps;
          await page.mouse.move(50 + (1230 - 50) * t, 50 + (750 - 50) * t);
          await page.waitForTimeout(dt);
        }
        // Sweep back diagonally for stress.
        for (let i = 0; i &lt;= steps; i++) {
          const t = i / steps;
          await page.mouse.move(1230 - (1230 - 50) * t, 50 + (750 - 50) * t);
          await page.waitForTimeout(dt);
        }

        const maxSep = await page.evaluate(() =&gt; (window as any).__blobDebug.maxAngularSeparation as number);
        // Acceptance criterion BR-02 from ROADMAP: max angular separation <= 8px.
        expect(maxSep).toBeLessThanOrEqual(8);
      });
    });

  cursor-mode acceptance; BR-03 (Plan 96-03) handles mobile.

  Do NOT use `reducedMotion: 'reduce'` here — we need cursor mode active.
