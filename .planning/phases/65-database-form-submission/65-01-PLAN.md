   ```typescript
   import { z } from 'zod';

   export const contactFormSchema = z.object({
     name: z
       .string()
       .min(2, 'Введите ваше имя'),
     phone: z
       .string()
       .regex(/^\+7\s?\(?\d{3}\)?\s?\d{3}[-\s]?\d{2}[-\s]?\d{2}$/, 'Введите корректный номер телефона'),
     specialization: z
       .string()
       .min(1, 'Выберите направление'),
     description: z
       .string()
       .optional()
       .default(''),
   });

   export type ContactFormInput = z.infer<typeof contactFormSchema>;
   ```
   The phone regex accepts the formatted output from ContactForm's `formatPhone()`: `+7 (XXX) XXX-XX-XX`. Also accepts variants without spaces/parens since we strip formatting before validation. The error messages match the existing Russian UI strings in ContactForm.tsx.

2. Create `next/src/lib/db/actions.ts`:
   ```typescript
   'use server';

   import { db } from './index';
   import { submissions } from './schema';
   import { contactFormSchema } from './validation';

   type ActionResult = {
     success: boolean;
     errors?: Record<string, string>;
   };

   export async function submitContactForm(formData: {
     name: string;
     phone: string;
     specialization: string;
     description: string;
     honeypot: string;
     loadTime: number;
   }): Promise<ActionResult> {
     // Spam check 1: honeypot field must be empty
     if (formData.honeypot) {
       return { success: true }; // Silent success for bots
     }

     // Spam check 2: submission must take at least 2 seconds
     if (Date.now() - formData.loadTime < 2000) {
       return { success: true }; // Silent success for bots
     }

     // Validate with Zod
     const result = contactFormSchema.safeParse({
       name: formData.name,
       phone: formData.phone,
       specialization: formData.specialization,
       description: formData.description,
     });

     if (!result.success) {
       const fieldErrors: Record<string, string> = {};
       for (const issue of result.error.issues) {
         const field = issue.path[0];
         if (typeof field === 'string') {
           fieldErrors[field] = issue.message;
         }
       }
       return { success: false, errors: fieldErrors };
     }

     // Map specialization values to human-readable Russian labels for DB storage
     const specMap: Record<string, string> = {
       consultation: 'Онлайн-консультация',
       treatment: 'Лечение за рубежом',
       checkup: 'Чек-ап',
       'not-sure': 'Пока не определился',
     };

     try {
       await db.insert(submissions).values({
         name: result.data.name.trim(),
         phone: result.data.phone,
         specialization: specMap[result.data.specialization] ?? result.data.specialization,
         description: result.data.description?.trim() || null,
       });

       return { success: true };
     } catch (error) {
       console.error('Failed to insert submission:', error);
       return {
         success: false,
         errors: { _form: 'Произошла ошибка. Пожалуйста, попробуйте позже.' },
       };
     }
   }
   ```

   Key design decisions:
   - The action receives plain data (not FormData) for type safety with the client component
   - Honeypot and timing checks happen BEFORE validation -- bots get a 200 success without any DB write
   - Zod validation runs server-side with the same error messages as client-side validation for consistency
   - The `_form` key in errors is a convention for form-level (non-field) errors
   - Specialization values are mapped from select option values to Russian labels before storage (same pattern as the original Directus flow)
   - `description` is stored as null when empty (not empty string) to match the original Directus nullable field

   async function handleSubmit(e: FormEvent) {
     e.preventDefault();
     setFormError('');

     // Client-side honeypot check (fast rejection, no network)
     if (website) {
       setFormState('success');
       return;

     // Client-side timing check (fast rejection, no network)
     if (Date.now() - loadTimeRef.current < 3000) {

     const validationErrors = validate();
     if (Object.keys(validationErrors).length > 0) {

       return;

     const result = await submitContactForm({

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> Server Action | Untrusted form data crosses from browser to server |
| Server Action -> PostgreSQL | Validated data written to database |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-65-01 | Spoofing | submitContactForm | mitigate | Server Action validates all input with Zod server-side; client-side validation is convenience only, never trusted |
| T-65-02 | Tampering | form data in transit | accept | Next.js Server Actions use POST with CSRF protection built-in; HTTPS in production (Phase 67) |
| T-65-03 | Information Disclosure | database errors | mitigate | catch block logs error server-side but returns generic "try again later" message to client; no stack traces exposed |
| T-65-04 | Denial of Service | form spam | mitigate | Honeypot field + timing check (2-second minimum) reject automated submissions; rate limiting deferred to reverse proxy (Phase 67 nginx) |
| T-65-05 | Injection | SQL injection via form fields | mitigate | Drizzle ORM uses parameterized queries; no raw SQL; Zod validates input shape before DB insert |
| T-65-06 | Elevation of Privilege | Server Action access | accept | Server Action is a public endpoint (form is public); no auth required for submission by design |
</threat_model>
