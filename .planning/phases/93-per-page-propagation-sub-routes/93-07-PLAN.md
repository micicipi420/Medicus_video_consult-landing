    test.describe('submission-e2e', () => {
      // collect record ids per spec for cleanup
      const createdIds: string[] = [];

      test.afterAll(async () => {
        if (createdIds.length === 0) return;
        const api = await pwRequest.newContext({ baseURL: DIRECTUS_URL });
        const auth = await api.post('/auth/login', { data: { email: ADMIN_EMAIL, password: ADMIN_PASSWORD } });
        const token = (await auth.json()).data.access_token;
        for (const id of createdIds) {
          await api.delete(`/items/submissions/${id}`, { headers: { Authorization: `Bearer ${token}` } }).catch(() => {});

      });

      for (const route of ROUTES) {
        test(`${route} — submission reaches Directus`, async ({ page }) => {
          const marker = `[E2E] ${route} ${Date.now()}`;
          await page.goto(route);
          await page.waitForLoadState('networkidle');

          // Locate the form (LeadFormSection on /checkup, /consultations, /treatment-abroad, /contacts)
          // Use accessible labels — Russian copy.
          await page.getByLabel(/имя/i).fill(marker);
          await page.getByLabel(/телефон/i).fill('+7 701 555 00 00');
          // adapt for specialization/dropdown if required field
          // adapt for description/textarea
          // ...

          // Submit
          await page.getByRole('button', { name: /отправить|оставить заявку/i }).click();

          // Wait for success indicator (ContactForm success overlay — see ContactForm.tsx)
          await expect(page.getByText(/спасибо|заявка принята|заявка отправлена/i)).toBeVisible({ timeout: 10_000 });

          // Verify in Directus via API
          const api = await pwRequest.newContext({ baseURL: DIRECTUS_URL });
          const auth = await api.post('/auth/login', { data: { email: ADMIN_EMAIL, password: ADMIN_PASSWORD } });
          const token = (await auth.json()).data.access_token;

          const res = await api.get(`/items/submissions?filter[name][_eq]=${encodeURIComponent(marker)}`, {
            headers: { Authorization: `Bearer ${token}` },
          });
          const body = await res.json();
          expect(body.data.length).toBeGreaterThanOrEqual(1);
          createdIds.push(body.data[0].id);
        });
      }
    });
