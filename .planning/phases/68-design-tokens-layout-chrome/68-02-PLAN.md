**HeaderClient.tsx -- rewrite the component:**

Replace the entire component. The header changes from `sticky top-0` solid white to `fixed top-4 left-4 right-4` glass pill.

```typescript
'use client';

import { cn } from '@/lib/utils';
import { useScrolled } from '@/hooks/use-scrolled';

export function HeaderClient({ children }: { children: React.ReactNode }) {
  const isScrolled = useScrolled();

  return (
    <header
      className={cn(
        'fixed z-50 transition-all duration-500 top-4 left-4 right-4 mx-auto max-w-7xl',
        'rounded-[2.5rem] px-4 md:px-8 border-[0.5px] border-white/50',
        'shadow-glass-header',
        'flex items-center',
        // Default glass state
        'bg-white/30 backdrop-blur-[40px] backdrop-saturate-[150%] py-5',
        // Scrolled glass state
        isScrolled && 'bg-white/50 backdrop-blur-[60px] backdrop-saturate-[180%] py-3',
      )}
    >
      {children}
    </header>
  );
}
```

Key differences from current:
- `fixed` instead of `sticky` (floats above content, layout.tsx main has pt-24 from Plan 01)
- `top-4 left-4 right-4` instead of `top-0` (floating with 16px margin)
- `mx-auto max-w-7xl` to center and constrain width on large screens
- `rounded-[2.5rem]` for pill shape
- `bg-white/30 backdrop-blur-[40px] backdrop-saturate-[150%]` glass background instead of solid `bg-white`
- `shadow-glass-header` instead of manual box-shadow
- `py-5` default, `py-3` when scrolled (padding reduction)
- `border-[0.5px] border-white/50` glass border
- Removed `h-16 md:h-[76px]` fixed height -- padding controls height now
- Scroll state adds `bg-white/50`, `backdrop-blur-[60px]`, `backdrop-saturate-[180%]`

**MobileMenu.tsx -- rewrite the component:**

Replace the entire component content. The mobile menu transitions from a white dropdown to a glass overlay panel.

```typescript
'use client';

import { useState, useEffect } from 'react';
import Link from 'next/link';
import { Menu, X, Phone } from 'lucide-react';
import { NAV_LINKS, PHONE_NUMBER, PHONE_DISPLAY } from '@/lib/navigation';

export function MobileMenu() {
  const [isOpen, setIsOpen] = useState(false);

  // Body scroll lock
  useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = 'hidden';
    } else {
      document.body.style.overflow = '';
    }
    return () => {
      document.body.style.overflow = '';
    };
  }, [isOpen]);

  return (
    <>
      {/* Burger button -- glass pill style, visible below lg */}
      <button
        onClick={() => setIsOpen(!isOpen)}
        aria-expanded={isOpen}
        aria-label={isOpen ? 'Закрыть меню' : 'Открыть меню'}
        className="lg:hidden p-2 text-mu-text-700 bg-white/50 rounded-full backdrop-blur-xl backdrop-saturate-[180%] border border-white/50"
      >
        {isOpen ? <X size={20} /> : <Menu size={20} />}
      </button>

      {isOpen && (
        <div className="fixed inset-0 z-40 lg:hidden">
          {/* Backdrop overlay */}
          <div
            className="absolute inset-0 bg-black/50"
            onClick={() => setIsOpen(false)}
          />

          {/* Glass nav panel */}
          <nav className="absolute top-24 left-4 right-4 bg-white/60 backdrop-blur-[80px] backdrop-saturate-[200%] shadow-glass-lg rounded-3xl overflow-hidden border-[0.5px] border-white/50">
            <div className="flex flex-col p-6 gap-2">
              {NAV_LINKS.map((link) =>
                link.href.startsWith('/') && !link.href.includes('#') ? (
                  <Link
                    key={link.href}
                    href={link.href}
                    onClick={() => setIsOpen(false)}
                    className="text-mu-text-900 hover:bg-white/40 rounded-2xl px-4 py-3 transition-colors font-medium tracking-tight"
                  >
                    {link.label}
                  </Link>
                ) : (
                  <a
                    key={link.href}
                    href={link.href}
                    onClick={() => setIsOpen(false)}
                    className="text-mu-text-900 hover:bg-white/40 rounded-2xl px-4 py-3 transition-colors font-medium tracking-tight"
                  >
                    {link.label}
                  </a>
                ),
              )}

              {/* Divider */}
              <div className="h-[0.5px] bg-white/40 my-2" />

              {/* Phone link with icon */}
              <a
                href={`tel:${PHONE_NUMBER}`}
                className="flex items-center gap-3 px-4 py-3 rounded-2xl hover:bg-white/40 transition-colors"
              >
                <Phone size={20} className="text-mu-blue" />
                <span className="text-mu-text-900 font-medium tracking-tight">
                  {PHONE_DISPLAY}
                </span>
              </a>

              {/* CTA button */}
              <a
                href="#contact"
                onClick={() => setIsOpen(false)}
                className="bg-gradient-to-r from-mu-blue to-mu-accent-blue text-white px-6 py-4 rounded-2xl font-extrabold tracking-tight shadow-lg mt-4 w-full text-center block"
              >
                Обсудить случай
              </a>
            </div>
          </nav>
        </div>
      )}
    </>
  );
}
```

Key differences from current:
- Burger button: `lg:hidden` instead of `md:hidden` (matches header breakpoint)
- Burger button styling: glass pill `bg-white/50 rounded-full backdrop-blur-xl backdrop-saturate-[180%] border border-white/50` instead of plain `p-2 -mr-2`
- Icon size: `20` instead of `24` (fits glass pill better)
- Overlay z-index: wrapper `fixed inset-0 z-40 lg:hidden` instead of separate backdrop
- Backdrop: `bg-black/50` instead of `bg-black/30`
- Panel: `top-24` instead of `top-16` (accounts for taller glass header), `bg-white/60 backdrop-blur-[80px] backdrop-saturate-[200%] shadow-glass-lg rounded-3xl border-[0.5px] border-white/50` instead of `bg-white shadow-xl rounded-2xl`
- Nav links: `text-mu-text-900 hover:bg-white/40 rounded-2xl` instead of `text-[#18212C] hover:bg-[#F0F7FF] rounded-xl`
- Added `font-medium tracking-tight` to all nav links
- Divider: `h-[0.5px] bg-white/40 my-2` instead of `<hr>` with `border-black/[0.06]`
- Phone: added `<Phone size={20} className="text-mu-blue" />` icon, `flex items-center gap-3` layout
- NEW: gradient CTA button "Obsudit sluchay" at bottom of panel -- full width, rounded-2xl, mt-4
- Link routing: same pattern as Header.tsx -- `/` paths use `<Link>`, `/#` paths use `<a>`
